cmake_minimum_required(VERSION 3.25)
project(CentipedePractice LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
foreach(configuration Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${configuration}" configuration_upper)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration_upper} "${CMAKE_CURRENT_SOURCE_DIR}/bin")
endforeach()

option(CENTIPEDE_BUILD_GAME "Build the graphical game (requires raylib)" ON)
add_library(centipede_core src/GameModel.cpp src/HighScores.cpp)
target_include_directories(centipede_core PUBLIC src)
if(MSVC)
    target_compile_options(centipede_core PRIVATE /W4)
else()
    target_compile_options(centipede_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(CENTIPEDE_BUILD_GAME)
    find_package(raylib 6.0 QUIET)
    if(NOT raylib_FOUND)
        include(FetchContent)
        set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(raylib
            GIT_REPOSITORY https://github.com/raysan5/raylib.git
            GIT_TAG 6.0
            GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(raylib)
    endif()
    add_executable(game src/main.cpp src/Renderer.cpp)
    target_link_libraries(game PRIVATE centipede_core raylib)
    if(MINGW)
        target_link_options(game PRIVATE -static-libgcc -static-libstdc++)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(tests tests/tests.cpp)
    target_link_libraries(tests PRIVATE centipede_core)
    add_test(NAME centipede_rules COMMAND tests)
endif()
